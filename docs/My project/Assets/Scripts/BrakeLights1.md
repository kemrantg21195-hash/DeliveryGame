### Пример 2. Гипотетическая структура MonoBehaviour (шаблон)

```csharp
using UnityEngine;

/// <summary>
/// Управление задними стоп-сигналами автомобиля.
/// </summary>
public class BrakeLights1 : MonoBehaviour
{
    [SerializeField] private Light[] brakeLights;   // ссылки на источники света
    [SerializeField] private Color  brakeColor = Color.red;

    private float _previousSpeed;

    private void Start()
    {
        SetLights(false);
    }

    private void Update()
    {
        float speed = GetVehicleSpeed();
        bool isBraking = speed < _previousSpeed - 0.1f;
        SetLights(isBraking);
        _previousSpeed = speed;
    }

    private float GetVehicleSpeed()
    {
        // Здесь обычно читается скорость Rigidbody
        return 0f;
    }

    private void SetLights(bool active)
    {
        foreach (var light in brakeLights)
        {
            if (light != null)
                light.enabled = active;
        }
    }
}
```
